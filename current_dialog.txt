function BibleBrowseDialog({ isOpen, onClose, index, availableBooks, onSelect }) {
    // Pages/Steps: 'testament', 'book', 'chapter', 'verse'
    const [step, setStep] = useState('testament');
    const [testament, setTestament] = useState(null); // 'OT' or 'NT'
    const [book, setBook] = useState(null);
    const [chapter, setChapter] = useState(null);
    const [sortDesc, setSortDesc] = useState(false);

    // Chunking state
    const [chapterGroupIndex, setChapterGroupIndex] = useState(0);
    const [verseGroupIndex, setVerseGroupIndex] = useState(0);

    // Reset when opened
    useEffect(() => {
        if (isOpen) {
            setStep('testament');
            setTestament(null);
            setBook(null);
            setChapter(null);
            setSortDesc(false);
            setChapterGroupIndex(0);
            setVerseGroupIndex(0);
        }
    }, [isOpen]);

    if (!isOpen) return null;

    const OT_BOOKS = availableBooks.slice(0, 39);
    const NT_BOOKS = availableBooks.slice(39);

    const handleTestamentSelect = (t) => {
        setTestament(t);
        setStep('book');
    };

    const handleBookSelect = (b) => {
        setBook(b);
        setChapterGroupIndex(0);
        setStep('chapter');
    };

    const handleChapterSelect = (c) => {
        setChapter(c);
        setVerseGroupIndex(0);

        // If we have index data and there's only 1 verse in this chapter, auto-select it
        if (index && index[book] && index[book][c]) {
            const verses = index[book][c];
            if (verses.length === 1) {
                onSelect(book, c, verses[0]);
                onClose();
                return;
            }
        }
        setStep('verse');
    };

    const handleVerseSelect = (v) => {
        onSelect(book, chapter, v);
        onClose();
    };

    const handleBack = () => {
        if (step === 'verse') setStep('chapter');
        else if (step === 'chapter') setStep('book');
        else if (step === 'book') setStep('testament');
    };

    // Prepare data for current view
    let displayBooks = [];
    if (step === 'book') {
        displayBooks = testament === 'OT' ? OT_BOOKS : NT_BOOKS;
        if (sortDesc) {
            displayBooks = [...displayBooks].sort((a, b) => BOOK_NAMES[a].localeCompare(BOOK_NAMES[b]));
        }
    }

    let chapterChunks = [];
    let currentChapterChunk = [];
    if (step === 'chapter' && index && index[book]) {
        let chapters = Object.keys(index[book]).map(Number);
        if (sortDesc) chapters.sort((a,b) => b - a);
        else chapters.sort((a,b) => a - b);

        for (let i = 0; i < chapters.length; i += 50) {
            chapterChunks.push(chapters.slice(i, i + 50));
        }
        currentChapterChunk = chapterChunks[chapterGroupIndex] || [];
    }

    let verseChunks = [];
    let currentVerseChunk = [];
    if (step === 'verse' && index && index[book] && index[book][chapter]) {
        let verses = index[book][chapter].map(Number);
        if (sortDesc) verses.sort((a,b) => b - a);
        else verses.sort((a,b) => a - b);

        for (let i = 0; i < verses.length; i += 50) {
            verseChunks.push(verses.slice(i, i + 50));
        }
        currentVerseChunk = verseChunks[verseGroupIndex] || [];
    }

    return (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-2 sm:p-4 md:p-10 bg-black/60 backdrop-blur-sm transition-opacity">
            <div className="bg-stone-50 w-full max-w-5xl h-[90vh] md:h-[80vh] rounded-2xl md:rounded-3xl shadow-2xl flex flex-col overflow-hidden relative border border-stone-200">

                {/* Header / Controls */}
                <div className="absolute top-0 inset-x-0 h-16 md:h-20 bg-gradient-to-b from-stone-200/80 to-transparent flex items-center justify-between px-4 md:px-8 z-10 pointer-events-none">
                    <div className="flex-1 pointer-events-auto flex items-center gap-2">
                        {step !== 'testament' && (
                            <button onClick={handleBack} className="hidden md:flex items-center gap-1 text-stone-500 hover:text-amber-700 font-bold text-xs uppercase tracking-wider bg-white/50 hover:bg-white px-3 py-1.5 rounded-full transition-all border border-stone-200 shadow-sm focus:outline-none focus:ring-2 focus:ring-amber-500">
                                <svg className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" /></svg>
                                Back
                            </button>
                        )}
                    </div>

                    <div className="flex-1 flex justify-center pointer-events-auto">
                        {step === 'book' && (
                            <button onClick={() => setSortDesc(!sortDesc)} className="flex items-center gap-1 text-stone-600 hover:text-amber-700 font-bold text-[10px] md:text-xs uppercase tracking-wider bg-white/80 hover:bg-white px-3 py-1.5 rounded-full transition-all border border-stone-200 shadow-sm focus:outline-none focus:ring-2 focus:ring-amber-500">
                                {sortDesc ? (<> <svg className="h-3 w-3" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M3 4h13M3 8h9m-9 4h6m4 0l4-4m0 0l4 4m-4-4v12" /></svg> A-Z </>) : (<> <svg className="h-3 w-3" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M3 4h13M3 8h9m-9 4h9m5-4v12m0 0l-4-4m4 4l4-4" /></svg> # </>)}
                            </button>
                        )}
                        {(step === 'chapter' || step === 'verse') && (
                            <button onClick={() => setSortDesc(!sortDesc)} className="flex items-center gap-1 text-stone-600 hover:text-amber-700 font-bold text-[10px] md:text-xs uppercase tracking-wider bg-white/80 hover:bg-white px-3 py-1.5 rounded-full transition-all border border-stone-200 shadow-sm focus:outline-none focus:ring-2 focus:ring-amber-500">
                                {sortDesc ? (<> <svg className="h-3 w-3" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 15l7-7 7 7" /></svg> DESC </>) : (<> <svg className="h-3 w-3" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" /></svg> ASC </>)}
                            </button>
                        )}
                    </div>

                    <div className="flex-1 flex justify-end pointer-events-auto">
                        <button onClick={onClose} className="p-2 text-stone-400 hover:text-stone-700 bg-white/50 hover:bg-white rounded-full transition-all border border-transparent hover:border-stone-200 shadow-sm focus:outline-none focus:ring-2 focus:ring-amber-500" aria-label="Close dialog">
                            <svg className="h-5 w-5 md:h-6 md:w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" /></svg>
                        </button>
                    </div>
                </div>

                {/* Main Content Area - Split Book Design */}
                <div className="flex-1 flex flex-col md:flex-row pt-14 md:pt-0">

                    {/* Mobile Back Button (Merged into content area for mobile) */}
                    {step !== 'testament' && (
                        <div className="md:hidden px-4 pt-2 pb-0">
                            <button onClick={handleBack} className="flex items-center gap-1 text-stone-500 font-bold text-xs uppercase tracking-wider focus:outline-none">
                                <svg className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" /></svg>
                                Back
                            </button>
                        </div>
                    )}

                    {/* Left Page (Visual only on desktop, merged on mobile) */}
                    <div className="hidden md:flex flex-1 flex-col p-8 border-r border-stone-300 bg-gradient-to-l from-stone-100/50 to-transparent shadow-[inset_-10px_0_15px_-10px_rgba(0,0,0,0.1)] overflow-y-auto">
                        {/* Page 1, 3, 5, 7 equivalent */}
                        <div className="flex-1 flex flex-col justify-center">
                            {step === 'testament' && (
                                <button onClick={() => handleTestamentSelect('OT')} className="w-full h-64 bg-amber-50 hover:bg-amber-100 border-2 border-amber-200 rounded-xl flex flex-col items-center justify-center gap-4 transition-all hover:scale-[1.02] hover:shadow-lg text-amber-900 group">
                                    <span className="text-4xl md:text-5xl font-serif font-black uppercase tracking-tight group-hover:text-amber-700 transition-colors">Old</span>
                                    <span className="text-2xl md:text-3xl font-serif italic text-amber-700/70">Testament</span>
                                </button>
                            )}
                            {step === 'book' && (
                                <div className="text-center space-y-4">
                                    <h2 className="text-3xl font-serif font-black uppercase tracking-tight text-stone-800">{testament === 'OT' ? 'Old Testament' : 'New Testament'}</h2>
                                    <p className="text-stone-500 italic font-serif">Select a book to continue.</p>
                                </div>
                            )}
                            {step === 'chapter' && (
                                <div className="text-center space-y-4">
                                    <h2 className="text-3xl font-serif font-black uppercase tracking-tight text-stone-800">{BOOK_NAMES[book]}</h2>
                                    <p className="text-stone-500 italic font-serif">Select a Chapter</p>
                                </div>
                            )}
                            {step === 'verse' && (
                                <div className="text-center space-y-4">
                                    <h2 className="text-3xl font-serif font-black uppercase tracking-tight text-stone-800">{BOOK_NAMES[book]} {chapter}</h2>
                                    <p className="text-stone-500 italic font-serif">Select a Verse to begin reading</p>
                                </div>
                            )}
                        </div>
                    </div>

                    {/* Right Page (Main interactive area) */}
                    <div className="flex-1 flex flex-col p-4 md:p-8 bg-gradient-to-r from-stone-100/50 to-transparent overflow-y-auto min-h-[50vh]">
                        {/* Mobile Title (Hidden on desktop) */}
                        <div className="md:hidden text-center mb-6 mt-10">
                            {step === 'testament' && <h2 className="text-2xl font-serif font-black uppercase text-stone-800">Select Testament</h2>}
                            {step === 'book' && <h2 className="text-2xl font-serif font-black uppercase text-stone-800">{testament === 'OT' ? 'Old Testament' : 'New Testament'}</h2>}
                            {step === 'chapter' && <h2 className="text-2xl font-serif font-black uppercase text-stone-800">{BOOK_NAMES[book]}</h2>}
                            {step === 'verse' && <h2 className="text-2xl font-serif font-black uppercase text-stone-800">{BOOK_NAMES[book]} {chapter}</h2>}
                        </div>

                        {step === 'testament' && (
                            <div className="flex-1 flex flex-col justify-center gap-4">
                                {/* Duplicate OT for mobile layout so they are stacked */}
                                <button onClick={() => handleTestamentSelect('OT')} className="md:hidden w-full h-40 bg-amber-50 hover:bg-amber-100 border-2 border-amber-200 rounded-xl flex flex-col items-center justify-center transition-all text-amber-900 shadow-sm">
                                    <span className="text-3xl font-serif font-black uppercase tracking-tight">Old</span>
                                    <span className="text-xl font-serif italic text-amber-700/70">Testament</span>
                                </button>
                                <button onClick={() => handleTestamentSelect('NT')} className="w-full h-40 md:h-64 bg-blue-50 hover:bg-blue-100 border-2 border-blue-200 rounded-xl flex flex-col items-center justify-center gap-2 md:gap-4 transition-all md:hover:scale-[1.02] hover:shadow-lg text-blue-900 group">
                                    <span className="text-3xl md:text-5xl font-serif font-black uppercase tracking-tight group-hover:text-blue-700 transition-colors">New</span>
                                    <span className="text-xl md:text-3xl font-serif italic text-blue-700/70">Testament</span>
                                </button>
                            </div>
                        )}

                        {step === 'book' && (
                            <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-4 gap-2 md:gap-3 content-start">
                                {displayBooks.map(b => (
                                    <button key={b} onClick={() => handleBookSelect(b)} className="bg-white border border-stone-200 hover:border-amber-400 hover:bg-amber-50 hover:shadow-md rounded-lg p-2 flex flex-col items-center justify-center transition-all focus:outline-none focus:ring-2 focus:ring-amber-500 group">
                                        <span className="text-lg md:text-xl font-black text-stone-800 group-hover:text-amber-700 uppercase">{b}</span>
                                        <span className="text-[9px] md:text-[10px] text-stone-500 truncate w-full text-center mt-1">{BOOK_NAMES[b]}</span>
                                    </button>
                                ))}
                            </div>
                        )}

                        {step === 'chapter' && (
                            <div className="flex flex-col h-full">
                                {chapterChunks.length > 1 && (
                                    <div className="flex flex-wrap gap-2 mb-4 justify-center">
                                        {chapterChunks.map((chunk, idx) => (
                                            <button key={idx} onClick={() => setChapterGroupIndex(idx)} className={`px-3 py-1 rounded-full text-xs font-bold transition-colors ${chapterGroupIndex === idx ? 'bg-amber-600 text-white' : 'bg-stone-200 text-stone-600 hover:bg-stone-300'}`}>
                                                {sortDesc ? `${chunk[0]}-${chunk[chunk.length-1]}` : `${chunk[0]}-${chunk[chunk.length-1]}`}
                                            </button>
                                        ))}
                                    </div>
                                )}
                                <div className="grid grid-cols-5 sm:grid-cols-6 md:grid-cols-5 gap-2 content-start">
                                    {currentChapterChunk.map(c => (
                                        <button key={c} onClick={() => handleChapterSelect(c)} className="bg-white border border-stone-200 hover:border-amber-400 hover:bg-amber-50 rounded-lg p-3 flex items-center justify-center font-bold text-stone-700 transition-all focus:outline-none focus:ring-2 focus:ring-amber-500">
                                            {c}
                                        </button>
                                    ))}
                                </div>
                            </div>
                        )}

                        {step === 'verse' && (
                            <div className="flex flex-col h-full">
                                {verseChunks.length > 1 && (
                                    <div className="flex flex-wrap gap-2 mb-4 justify-center">
                                        {verseChunks.map((chunk, idx) => (
                                            <button key={idx} onClick={() => setVerseGroupIndex(idx)} className={`px-3 py-1 rounded-full text-xs font-bold transition-colors ${verseGroupIndex === idx ? 'bg-amber-600 text-white' : 'bg-stone-200 text-stone-600 hover:bg-stone-300'}`}>
                                                {sortDesc ? `${chunk[0]}-${chunk[chunk.length-1]}` : `${chunk[0]}-${chunk[chunk.length-1]}`}
                                            </button>
                                        ))}
                                    </div>
                                )}
                                <div className="grid grid-cols-5 sm:grid-cols-6 md:grid-cols-5 gap-2 content-start">
                                    {currentVerseChunk.map(v => (
                                        <button key={v} onClick={() => handleVerseSelect(v)} className="bg-white border border-stone-200 hover:border-amber-400 hover:bg-amber-50 rounded-lg p-3 flex items-center justify-center font-bold text-stone-700 transition-all focus:outline-none focus:ring-2 focus:ring-amber-500">
                                            {v}
                                        </button>
                                    ))}
                                </div>
                            </div>
                        )}

                    </div>
                </div>
            </div>
        </div>
    );
}

function App() {

  const [index, setIndex] = useState(null); // { GEN: { 1: [days], ... }, ... }
  const [availableBooks, setAvailableBooks] = useState([]);

  const [selectedBook, setSelectedBook] = useState(null);
  const [selectedChapter, setSelectedChapter] = useState(null);
